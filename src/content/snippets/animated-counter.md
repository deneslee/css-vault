---
title: "Animated Number Counter"
category: "Text & Numbers"
tags: ["counter", "@property", "keyframes", "numbers"]
controls:
  - label: "Count to"
    property: "--count-to"
    reactProp: "to"
    primary: true
    type: "range"
    default: 100
    min: 10
    max: 1000
    step: 10
  - label: "Duration"
    property: "--count-duration"
    reactProp: "duration"
    type: "range"
    default: 2
    min: 0.5
    max: 5
    step: 0.1
    unit: "s"
cssCode: |
  /* Registering --num as an integer is what makes it animatable:
     unregistered custom properties just flip at 50%. */
  @property --num {
    syntax: "<integer>";
    initial-value: 0;
    inherits: false;
  }

  .stat {
    font: 700 3rem/1 system-ui, sans-serif;
    font-variant-numeric: tabular-nums;
    counter-reset: n var(--num);
    animation: count var(--count-duration, 2s) ease-out forwards;
  }

  /* The number is drawn by CSS: the counter holds --num's current value. */
  .stat::after {
    content: counter(n);
  }

  @keyframes count {
    to { --num: var(--count-to, 100); }
  }
htmlCode: |
  <div class="stat" role="img" aria-label="100"></div>
reactCode: |
  // AnimatedCounter.tsx — put this snippet's CSS in AnimatedCounter.css next to this file
  import type { CSSProperties } from 'react';
  import './AnimatedCounter.css';

  interface AnimatedCounterProps {
    /** Whole number to count up to. */
    to?: number;
    /** Seconds the count takes. */
    duration?: number;
  }

  export function AnimatedCounter({ to = 100, duration = 2 }: AnimatedCounterProps) {
    return (
      <div
        // A new key remounts the element, so the count replays when `to` changes.
        key={to}
        className="stat"
        role="img"
        aria-label={String(to)}
        style={{
          '--count-to': to,
          '--count-duration': `${duration}s`,
        } as CSSProperties}
      />
    );
  }
---

### Implementation notes

A count-up with no JavaScript. `@property` registers `--num` as an `<integer>`, so the browser can interpolate it from 0 to the target. `counter-reset` copies its current value into a CSS counter each frame, and `::after` prints it with `content: counter(n)`.

1. Without `@property` the custom property can't animate: it would jump straight to the final number.
2. `forwards` keeps the final number on screen once the count ends.
3. The target comes from `--count-to`, so one set of keyframes works for any number.

Text generated by `content` isn't reliably read by screen readers, so the element carries the final value as its `aria-label`.
